from dab4pt.store.orders;

select order_id
from dab4pt.store.orders

-- Classify quantity ordered as:
--single (1 order), small (2-10 orders), bulk (more than 10).
select order_id, quantity,
case
when quantity=1 then 'single'
when quantity between 2 and 10 then 'small'
else 'bulk' end as status
from dab4pt.store.orders;

select order_id,
case
when quantity=1 then 'single'
when quantity between 2 and 10 then 'small'
else 'bulk' end as status ,quantity
from dab4pt.store.orders;

--show the order id ,profit and a new column called profit_level
--based on the following:
-- if the profit is less than 0 then 'Negative',
--if it is equal to 0 then 'Breakeven',
-- else 'Positive '
select order_id, profit,
CASE
WHEN profit<0 then 'Negative'
When profit=0 then 'Break-even'
else 'Positive'
End as profit_level
from dab4pt.store.orders;


-- Show customer_id, ship_mode, a column called speed
-- that is equal to 'Fast' if the ship_mode is 'Same Day'and 'First Class', otherwise 'Slow'.
SELECT customer_id, ship_mode,
CASE
when ship_mode in ('Same Day','First Class') then 'Fast'
else 'Slow'
End as speed
from dab4pt.store.orders;
