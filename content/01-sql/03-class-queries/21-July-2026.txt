-- show product id and product name columns from products table
SELECT  product_id, product_name
from dab4pt.store.products;

-- Show countries , regions and salesperson.
Select country, region, salesperson
From dab4pt.store.regions;

--Show distinct Salesperson
SELECT distinct salesperson
from dab4pt.store.regions;

-- show me unique values of category column from products table
select distinct category
from dab4pt.store.products;

-- How many unique products do we have?
select distinct product_id
from dab4pt.store.products;


-- show me the first 10 transactions of order table, sorted by profit decending order
SELECT *
from dab4pt.store.orders
order by profit desc
limit 10;

-- show me first 20 order_id, ship_date,sales sorted by sales decending order
select order_id, ship_date, sales
from dab4pt.store.orders
order by profit desc
limit 20;


--List order_ids with the lowest profit. show lowest 15
SELECT order_id
FROM dab4pt.store.orders
order by profit asc
limit 15;


--Display all orders sorted by order date from newest to oldest.
SELECT *
FROM dab4pt.store.orders
order by order_date desc;
