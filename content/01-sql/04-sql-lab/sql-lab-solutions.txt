/*-------------------------------------------------------------------------------------------------
Q21: Find products that were ordered more than 100 times but returned fewer than 5 times. (2811 rows)
-------------------------------------------------------------------------------------------------*/
SELECT p.product_id, p.product_name,
  COUNT(DISTINCT o.order_id) AS order_count,
  COUNT(DISTINCT r.order_id) AS return_count
FROM `dsb-pt-123456.Super_store.Products` p
INNER JOIN `dsb-pt-123456.Super_store.Orders` o
  ON p.product_id = o.product_id
LEFT JOIN `dsb-pt-123456.Super_store.Returns` r
  ON p.product_id = r.product_id
GROUP BY p.product_id, p.product_name
HAVING COUNT(DISTINCT o.order_id) > 100
   AND COUNT(DISTINCT r.order_id) < 5
Order by 3 DESC;
